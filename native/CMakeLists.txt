cmake_minimum_required(VERSION 3.25)
project(leafseamer_clock LANGUAGES C CXX)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "unsupported_clock_platform")
endif()
if(NOT EXISTS "${NODE_INCLUDE_DIR}/node_api.h")
  message(FATAL_ERROR "node_headers_required")
endif()
add_library(leafseamer_clock SHARED clock.cc)
target_include_directories(leafseamer_clock PRIVATE "${NODE_INCLUDE_DIR}")
target_compile_definitions(leafseamer_clock PRIVATE NAPI_VERSION=8 BUILDING_NODE_EXTENSION)
target_compile_features(leafseamer_clock PRIVATE cxx_std_17)
set_target_properties(leafseamer_clock PROPERTIES PREFIX "" SUFFIX ".node")
if(MSVC)
  target_compile_definitions(leafseamer_clock PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(leafseamer_clock PRIVATE /W4 /WX /utf-8 /guard:cf /Brepro)
  target_link_options(leafseamer_clock PRIVATE /guard:cf /Brepro)
  set_property(TARGET leafseamer_clock PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
  target_link_libraries(leafseamer_clock PRIVATE "${NODE_LIBRARY}" mincore powrprof)
else()
  target_compile_options(leafseamer_clock PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
  target_link_options(leafseamer_clock PRIVATE -Wl,-z,relro,-z,now)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/compiler.json" CONTENT "{\"id\":\"${CMAKE_CXX_COMPILER_ID}\",\"version\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"system\":\"${CMAKE_SYSTEM_NAME}\"}\n")
if(WIN32)
  if(NOT EXISTS "${SQLITE_SOURCE_DIR}/sqlite3.c")
    message(FATAL_ERROR "sqlite_source_required")
  endif()
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/process-journal.sql" JOURNAL_SQL)
  file(WRITE "${CMAKE_BINARY_DIR}/process-journal-sql.h" "#pragma once\nstatic constexpr const char* processJournalSql = R\"SQL(${JOURNAL_SQL})SQL\";\n")
  add_library(leafseamer_journal_sqlite STATIC "${SQLITE_SOURCE_DIR}/sqlite3.c")
  target_compile_definitions(leafseamer_journal_sqlite PRIVATE SQLITE_THREADSAFE=0 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
  set_property(TARGET leafseamer_journal_sqlite PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
  add_executable(leafseamer_process_host process-host.cc)
  target_include_directories(leafseamer_process_host PRIVATE "${SQLITE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}")
  target_compile_features(leafseamer_process_host PRIVATE cxx_std_17)
  target_compile_definitions(leafseamer_process_host PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(leafseamer_process_host PRIVATE /W4 /WX /utf-8 /guard:cf /Brepro)
  target_link_options(leafseamer_process_host PRIVATE /guard:cf /Brepro)
  set_property(TARGET leafseamer_process_host PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
  target_link_libraries(leafseamer_process_host PRIVATE bcrypt advapi32 leafseamer_journal_sqlite)
endif()
